// Copy of ntsc-adaptive-final.glsl carrying ONE change, so it can be A/B'd
// against the original: the dot-crawl state count is now resolved from the
// source instead of being a fixed 4. See "ntsc_crawl_states" under "New in
// -final" below and fragment_resolve_crawl_states(). Everything else in this
// file is byte-identical to ntsc-adaptive-final.glsl.
//
// Derived from ntsc-adaptive-base.glsl, the raw mpv-libretro conversion of
// guest's advanced NTSC passes from the libretro slang-shaders repo. Every
// divergence from -base is marked "PATCHED:" at its site; this is the index.
//
// Carried over from ntsc-adaptive-crawl and verified:
//
//   fragment_mix_mat  Was a file-scope initializer reading BRIGHTNESS /
//                     FRINGING / ARTIFACTING / SATURATION, which vertex_main()
//                     only assigns later. Global initializers run before
//                     hook() calls vertex_main(), so the matrix was built from
//                     zeros and chroma was annihilated -- a black and white
//                     picture. Now assigned at the top of fragment_main().
//                     This is the single most important conversion fix here.
//
//   quality           Restored as a preset selector (0 = S-Video, 1 =
//                     Composite, 2 = RF), which is all upstream's three
//                     .slangp presets differ by. The advanced pass had
//                     demoted it to a label with min == max == 0.
//                     quality = -1 hands control back to cust_artifacting /
//                     cust_fringing.
//
//   ntsc_ref_width    Subcarrier calibration. mod_phase advances by
//                     pix_no.x * CHROMA_MOD_FREQ, so the picture carries
//                     OriginalSize.x * res * 0.5333 subcarrier cycles. Real
//                     NTSC is ~188 across the active line, which needs
//                     OriginalSize.x * res == 353. Upstream assumes a ~320px
//                     console source and leaves res at 1, which on a wide
//                     frame yields a subcarrier several MHz too high.
//                     NOTE: with analog-res-final at analog_width = 352 the
//                     ratio is 353/352 and the min(..., 1.) clamp makes this a
//                     no-op -- the two shaders are already in agreement. It
//                     only starts doing work if analog_width changes.
//
//   res_lines         The calibration above is HORIZONTAL only. pix_no.y is
//                     the scanline index and mod(pix_no.y, 2.) is what flips
//                     chroma phase per line, which needs a step of exactly 1
//                     per texture row. Folding the calibration into `res`
//                     broke that. res_lines keeps an uncalibrated copy.
//
//   crawl_field       Four-state travelling dot crawl. NTSC's colour frame is
//                     four fields (227.5 cycles/line puts adjacent lines 180
//                     deg apart and adjacent fields 90 deg apart), so the
//                     4-state 90 deg rotation is the physically correct
//                     sequence FOR AN INTERLACED RASTER; upstream's
//                     mod(FrameCount, 2.) only inverts in place.
//                     ntsc_crawl_states = 2 reproduces upstream.
//                     Superseded in this copy by the Auto resolution below -
//                     four states is right for 480i and wrong for 240p.
//
//   merge_field       The field-merge partner must sit half a cycle from
//                     crawl_field or the average does not cancel the
//                     modulation. Upstream used FrameCount + 1, which is the
//                     opposite field only when there are 2 states; at 4 states
//                     it landed on the same value at n%4==2 and 90 deg away at
//                     n%4==1,3. Offsetting by states/2 is correct for any
//                     state count and collapses to mod(n+1, 2) at 2 states.
//                     Dormant under the current retro.conf, which sets
//                     ntsc_fields=0 (merge off) to keep the dot crawl.
//
//   ntsc_frame_div    MUST BE 1 in this chain. The subcarrier is 455/2 times
//                     the line rate, so a line carries exactly 227.5 cycles
//                     and a 262.5-line field carries 59718.75 -- the fraction
//                     0.75 is a 270 deg (= -90 deg) phase step per FIELD, and
//                     four fields return to identity. So the crawl advances
//                     one 90 deg step per field, and with vf fps=60000/1001
//                     feeding crt-royale's bob, one mpv frame IS one field.
//                     ntsc_frame_div must therefore match royale_frame_div in
//                     crt-royale-final; the two count the same thing. At 2 the
//                     crawl runs at half rate (7.5 Hz instead of 15 Hz) and,
//                     worse, both fields of a source frame get the SAME chroma
//                     phase when they should sit 90 deg apart.
//
//   final alpha       ntsc-pass3 writes a scanline weight into alpha, which
//                     RetroArch discards on the final pass but mpv composites.
//                     Forced to 1.0 in the LAST pass only; earlier passes
//                     really do carry signal in alpha (pass3 reads NPass1.a).
//
// New in -final:
//
//   ntsc_line_period  Rows of the input texture per NTSC scanline. See the
//                     PATCHED note at the chroma_phase site. AUTO (-1) by
//                     default; 1 is exactly upstream behaviour.
//
//                     THE RULE IS ABOUT THE DISPLAY, NOT THE SOURCE: it is 2
//                     whenever crt-royale's is_interlaced() fires and 1 when it
//                     does not. What matters is only that something downstream
//                     shows every OTHER texture row as one field. It is
//                     irrelevant whether the rows got interleaved by a real
//                     interlaced capture or by crt-royale synthesising fields
//                     from a progressive frame - the phase must alternate every
//                     two rows either way. So:
//
//                       240p console (< 288.5 lines)      -> 1
//                       480i / 576i                       -> 2
//                       1080p driven as a 1080i sim       -> 2
//                       interlace_detect_toggle = 0       -> 1
//
//                     Upstream's default of 1 is right only because upstream
//                     targets 240p, where one texture row IS one displayed
//                     scanline. Since the deciding factor is a row count this
//                     pass can measure for itself, Auto resolves it from
//                     OriginalSize.y rather than making it something to keep in
//                     sync by hand - see fragment_resolve_line_period(), which
//                     also documents the one case Auto cannot see.
//
//   ntsc_crawl_states Now AUTO (-1) by default instead of a fixed 4, because
//                     the number of chroma-phase states is a property of the
//                     raster, not a preference:
//
//                       480i / bob-fielded (line_period 2)    -> 4, travelling
//                       240p progressive, 2-phase             -> 1, FROZEN
//                       240p progressive, 3-phase (NES-class) -> 3, rotating
//                       263-line 240p                         -> 2, set by hand
//
//                     A 262-line 240p frame carries a whole number of
//                     subcarrier cycles (227.5 * 262 = 59605.00), so the
//                     pattern never moves; the .5 line of interlace is what
//                     leaves .75 of a cycle over per field and makes 480i
//                     crawl. Running four states on a 240p pipeline invents a
//                     15 Hz colour shimmer no 240p console ever produced.
//                     4 restores the previous behaviour exactly.
//
//                     crawl_field is also now scaled per phase branch
//                     (crawl_unit): the 3-phase branch multiplies it by 120 deg
//                     rather than 180, so N states only ever meant N evenly
//                     spaced phases in the 2-phase branch. No-op for 2-phase,
//                     which is everything this chain runs.
//
//                     The rainbow pass in ntsc-pass3 carries its own copy of
//                     the resolution and freezes its field-parity line shift
//                     when the crawl is frozen. Dormant at ntsc_rainbow1 = 0.
//
//   The dead "store viewport size" pass is deleted -- nothing binds VIEWPORT
//   (`target_size` is a libplacebo builtin), and HEIGHT OUTPUT.width made
//   it a viewport-width-squared texture, rendered every frame for nothing.
//
// Reviewed and left as upstream:
//
//   ntsc_phase = 1 (Auto) resolves to 2-phase for any source wider than 300px,
//   which is the right model for real NTSC: 227.5 subcarrier cycles per line
//   means consecutive lines are 180 deg apart.
//
// NOTE ON COMMENTS IN THIS FILE
// libplacebo locates the start of the shader with a RAW SUBSTRING search for
// the directive prefix -- pl_str_find() in custom_mpv.c, not anchored to the
// start of a line. Writing that three-character prefix (slash slash bang)
// anywhere in prose, even mid-sentence inside a comment, makes the parser begin
// parsing there and mangles or rejects the whole file. So directives are named
// bare in these comments: WIDTH, HEIGHT, PARAM, HOOK, never with the prefix.
// ===========================================================================

void vertex_main() {
    // PATCHED: subcarrier calibration.
    // pix_no.x = vTexCoord.x * OriginalSize.x * res * 4, and mod_phase advances
    // by pix_no.x * CHROMA_MOD_FREQ, so the number of subcarrier cycles across
    // the picture is OriginalSize.x * res * 0.5333. Real NTSC puts ~188 cycles
    // across the active line, which needs OriginalSize.x * res == 353.
    // Upstream assumes a ~320px console source and leaves res at 1. On a
    // 1920-wide frame that yields 1024 cycles - a 19.5 MHz subcarrier. Scaling
    // res by 353/OriginalSize.x pins it to 3.58 MHz at any input resolution.
    float res = min(params.ntsc_scale * vertex_auto_rez * (ntsc_ref_width / params.OriginalSize.x), 1.);
    // The calibration is a HORIZONTAL correction only. pix_no.y is the scanline
    // index - mod(pix_no.y, 2.) is what inverts the chroma phase every line, and
    // that only works if it steps by exactly 1 per texture row. Upstream writes
    // the y term as res/auto_rez precisely so auto_rez cancels out. Folding the
    // subcarrier calibration into res broke that: at 768 wide it stepped 0.46 per
    // row, smearing the per-line alternation over 4.35 rows and flattening the
    // dot structure. Keep an uncalibrated copy for the line index.
    float res_lines = min(params.ntsc_scale * vertex_auto_rez, 1.);
    float OriginalSize = params.OriginalSize.x * vertex_auto_rez;
    vertex_gl_Position = global.MVP * Position;
    vTexCoord = TexCoord * vec2(params.speedup, 1.);
    pix_no = vTexCoord * params.OriginalSize.xy * vec2(res, res_lines / vertex_auto_rez) * vec2(4., 1.);
    phase = params.ntsc_phase < 1.5 ? OriginalSize > 300. ? 2. : 3. : params.ntsc_phase > 2.5 ? 3. : 2.;
    if (params.ntsc_phase == 4.) phase = 3.; else if (params.ntsc_phase == 5.) phase = 2.;
    CHROMA_MOD_FREQ = phase == 2. && params.ntsc_phase != 5. ? 4. * 3.1415927 / 15. : OriginalSize <= 300. || phase == 3. ? 3.1415927 / 3. : 3.1415927 / 2.;
    // PATCHED: re-wire `quality` as a preset selector.
    // Guest's *advanced* NTSC pass (which this file was generated from) demoted
    // `quality` to a label-only parameter - upstream declares it with
    // min == max == 0 and never reads it; the "A&F Values" text is just a hint
    // telling you what to type into cust_artifacting/cust_fringing by hand.
    // (Upstream's Svideo/Composite/RF .slangp presets differ in nothing but
    // those two values: 0/0, 1/1, 2/2.) The older ntsc-adaptive pass DID derive
    // A&F from quality, which is the behaviour this restores. quality = -1
    // hands control back to the cust_* values.
    ARTIFACTING = params.quality > -0.5 ? params.quality : params.cust_artifacting;
    FRINGING    = params.quality > -0.5 ? params.quality : params.cust_fringing;
    SATURATION = params.ntsc_sat;
    BRIGHTNESS = params.ntsc_bright;
    MERGE = 0.;
    if (params.ntsc_fields == -1. && phase == 3.) MERGE = 1.; else if (params.ntsc_fields == 0.) MERGE = 0.; else if (params.ntsc_fields == 1.) MERGE = 1.;
}

float fragment_get_luma(vec3 c) {
    return dot(c, vec3(0.2989, 0.587, 0.114));
}

float fragment_smothstep(float e0, float e1, float x) {
    return clamp((x - e0) / (e1 - e0), 0., 1.);
}

// PATCHED: resolve ntsc_line_period. At the Auto default it mirrors the height
// test in crt-royale's is_interlaced(), which is the thing that decides whether
// every other row becomes a field. All three NTSC passes preserve height (each
// HEIGHT directive is the previous pass's height times 1), so the row count
// crt-royale tests is exactly this pass's OriginalSize.y and no communication
// between the two files is needed.
//
// The 1080i arm is unconditional here, whereas crt-royale gates it behind
// interlace_1080i, and crt-royale can also be switched off wholesale with
// interlace_detect_toggle. Those two are in the other file and mpv gives each
// shader its own parameter namespace, so Auto cannot see them. They are static
// config rather than per-file properties, so the escape hatch is the override:
// if interlace_detect_toggle = 0, or interlace_1080i = 0 on a 1080-line source,
// set ntsc_line_period = 1 explicitly.
float fragment_resolve_line_period() {
    if (ntsc_line_period > 0.5) return ntsc_line_period;
    float lines = params.OriginalSize.y;
    bool sd = lines > 288.5 && lines < 576.5;
    bool hd = lines > 1079.5 && lines < 1080.5;
    return sd || hd ? 2. : 1.;
}

// PATCHED: resolve how many distinct temporal chroma states the source has.
// The count is not a taste knob, it is arithmetic: the subcarrier runs at
// 227.5 cycles per line, so multiplying by the lines in one vertical period
// and keeping the FRACTION gives the phase step from one displayed field to
// the next. Only the fraction matters, because a whole cycle is identity.
//
//   480i field   262.5 lines  59718.75 cycles  frac .75  -90 deg   -> 4 states
//   240p frame   262   lines  59605.00 cycles  frac .00    0 deg   -> 1, FROZEN
//   240p frame   263   lines  59832.50 cycles  frac .50  180 deg   -> 2 states
//
// A 240p console emits a WHOLE number of subcarrier cycles per frame, so every
// frame starts the subcarrier at the same phase and the dot pattern does not
// move at all. This is the origin of the "240p has no dot crawl" folklore, and
// it is literally true for the 262-line case: the checkerboard is nailed to the
// picture. Crawl exists in 480i precisely BECAUSE of the half line - the extra
// .5 line per field is what leaves .75 of a cycle over.
//
// Before this patch the crawl ran a fixed 4-state rotation whatever the source
// was, so a 240p pipeline cycled through four chroma phases that a real 240p
// console never produces - a 15 Hz colour shimmer on saturated edges with no
// physical counterpart. Auto now keys off the same resolved line_period that
// the per-line phase flip uses, which is exactly "is crt-royale bob-fielding
// this into fields": bob-fielding means a half-line offset exists, means 4.
//
// The 3-phase branch is a separate axis and is NOT about interlacing. A
// 3-phase source (NES: 341 dots/line, 56.8333 cycles/line, 262 lines ->
// 14890.33 cycles/frame) leaves a third of a cycle over per frame, so it
// rotates 120 deg per frame even though it is progressive - that is what the
// 3-phase artifact model is FOR. So Auto gives it 3 states, not 1.
//
// Overrides: 263-line 240p sources want an explicit 2. The pre-patch behaviour
// is ntsc_crawl_states = 4, and 1 freezes the pattern outright.
float fragment_resolve_crawl_states(float line_period) {
    if (ntsc_crawl_states > 0.5) return ntsc_crawl_states;
    if (line_period > 1.5) return 4.;   // bob-fielded: half-line offset exists
    return phase > 2.5 ? 3. : 1.;       // progressive: 3-phase rotates, 2-phase is frozen
}

void fragment_main() {
    fragment_mix_mat = mat3(BRIGHTNESS, FRINGING, FRINGING, ARTIFACTING, 2. * SATURATION, 0., ARTIFACTING, 0., 2. * SATURATION);
    if (vTexCoord.x > 1.) FragColor = 0..xxxx; else {
        float res = params.ntsc_scale * fragment_auto_rez * (ntsc_ref_width / params.OriginalSize.x);
        float mit = params.ntsc_taps;
        if (params.ntsc_charp > 0.25 && phase == 2.) mit = clamp(mit, 8., min(params.ntsc_taps, 14.));
        mit = fragment_smothstep(16., 8., mit) * 0.325;
        vec3 col = texture(Source, vTexCoord).rgb;
        vec3 yiq = fragment_rgb2yiq(col);
        yiq.x = pow(yiq.x, params.ntsc_gamma);
        float lum = yiq.x;
        if (params.ntsc_phase == 4.) {
            vec2 dx = vec2(params.OriginalSize.z / fragment_auto_rez, 0.);
            float c1 = fragment_get_luma(texture(Source, vTexCoord - dx).rgb);
            float c2 = fragment_get_luma(texture(Source, vTexCoord + dx).rgb);
            float mix01 = min(5. * abs(c1 - c2), 1.);
            c1 = pow(c1, params.ntsc_gamma);
            c2 = pow(c2, params.ntsc_gamma);
            yiq.x = mix(min(0.5 * (yiq.x + max(c1, c2)), max(yiq.x, min(c1, c2))), yiq.x, mix01);
        }
        vec3 tmp = yiq;
        vec3 yiq2 = yiq;
        vec3 yiqs = yiq;
        vec3 yiqz = yiq;
        // PATCHED: the field-merge partner must track the crawl state machine.
        // Merging works by averaging this field against one whose chroma is
        // HALF A CYCLE away, so the modulation cancels and the dot pattern goes
        // static. Upstream got that partner with FrameCount+1: at 2 states, the
        // next field IS the opposite one, so "next" and "opposite" coincided.
        // The crawl patch moved the main term below to a 4-state travelling
        // sequence but left this one on the legacy 2-state form, so they stopped
        // being complementary - at n%4==2 the two terms landed on the SAME value
        // and 0.5*(yiq+yiq2) became a no-op, at n%4==1,3 they sat 90 deg apart
        // and smeared hue instead of cancelling. Only n%4==0 still cancelled.
        // Offsetting by states/2 restores "opposite phase" for any state count,
        // and collapses back to upstream's mod(n+1,2) exactly when states == 2.
        // PATCHED: NTSC scanline index for the per-line chroma phase flip.
        // Upstream flips phase every texture ROW, which is right only when a row
        // IS a displayed scanline (240p). It is wrong whenever crt-royale is
        // bob-fielding, because then rows n and n+2 are consecutive lines of the
        // SAME field and rows n, n+1 belong to different fields. In a real
        // interlaced signal the 180 deg flip is per TRANSMITTED line, and field
        // A's transmitted lines k = 0,1,2... are frame rows 0,2,4..., so the
        // phase of row r is 180 deg * (r/2) - alternating every two rows.
        //
        // This has nothing to do with how the rows came to be interleaved. It is
        // equally true of a woven 480i capture and of a progressive 1080p frame
        // that crt-royale is splitting into synthetic fields for a 1080i sim.
        //
        // Flipping per row instead lands every line of the displayed field on
        // the SAME phase, so the dot pattern degenerates from a checkerboard
        // into vertical stripes - and since the other field sits at the opposite
        // phase, those stripes invert at field rate: a fixed vertical comb on
        // saturated edges flickering at 30 Hz. The 3-scanline vertical beam
        // average in crt-royale reinforces it rather than cancelling it.
        // ntsc_line_period = 2 flips every two rows instead, so phase still
        // alternates line to line within the field that actually gets shown.
        // The 90 deg field-to-field offset is already handled temporally by
        // crawl_field, so nothing extra is needed for the other field.
        // ntsc_line_period = 1 reproduces upstream exactly.
        //
        // Note this divides pix_no.y, which steps 1 per ORIGINAL row only while
        // res_lines / auto_rez == 1, i.e. at ntsc_scale = 1 with auto_res off.
        // Lower ntsc_scale and the row pairing stops lining up with real row
        // pairs; that is pre-existing and unrelated to the Auto detect.
        float line_period = fragment_resolve_line_period();
        float line_idx = line_period > 1.5
            ? floor(pix_no.y / line_period) + 0.5
            : pix_no.y;
        // PATCHED: state count is resolved from the source instead of being a
        // fixed 4, so a progressive 240p pipeline gets the frozen pattern a real
        // 240p console produces. See fragment_resolve_crawl_states().
        //
        // PATCHED: crawl_field is scaled into the units of whichever branch
        // consumes it. crawl_field is added INSIDE the chroma_phase expression,
        // where 2-phase multiplies by pi (one unit = 180 deg, a full turn = 2
        // units) and 3-phase multiplies by 2pi/3 (one unit = 120 deg, a full
        // turn = 3 units). The old code hardcoded the 2-phase scale in both, so
        // asking for N states only actually produced N evenly spaced phases in
        // the 2-phase branch; 3-phase got N steps of 240/N deg, which lands on
        // the correct 120 deg rotation for exactly one value of N and not the
        // one Auto wants. crawl_unit fixes the scale per branch and is a no-op
        // for every 2-phase configuration, which is everything in this chain -
        // Auto picks 2-phase for any source wider than 300 px.
        //
        // Consequence for the 3-phase branch: ntsc_crawl_states = 2 is now a
        // 0/180 deg flip there rather than upstream's 0/120, since 0/120 was
        // never 2 evenly spaced states in the first place. Upstream's exact
        // 3-phase behaviour is no longer reachable by a state count; Auto's 3
        // (0/120/240) is the sequence a real 3-phase console emits.
        float crawl_states = fragment_resolve_crawl_states(line_period);
        float crawl_unit   = phase < 2.5 ? 2. : 3.;
        // PATCHED: ntsc_frame_hold >= 0 pins the dot-crawl phase so a paused
        // re-render cannot advance it. See royale_frame_hold.
        float crawl_n     = ntsc_frame_hold >= 0. ? ntsc_frame_hold
                          : floor(fragment_FrameCount / ntsc_frame_div);
        float crawl_field = (crawl_unit / crawl_states) * mod(ntsc_crawl_dir * crawl_n, crawl_states);
        float merge_field = (crawl_unit / crawl_states) * mod(ntsc_crawl_dir * crawl_n + 0.5 * crawl_states, crawl_states);
        if (MERGE > 0.5) {
            float chroma_phase2 = phase < 2.5 ? 3.1415927 * (mod(line_idx, 2.) + merge_field) : 0.6667 * 3.1415927 * (mod(line_idx, 3.) + merge_field);
            float mod_phase2 = chroma_phase2 + pix_no.x * CHROMA_MOD_FREQ;
            float i_mod2 = cos(mod_phase2);
            float q_mod2 = sin(mod_phase2);
            yiq2.yz *= vec2(i_mod2, q_mod2);
            yiq2 *= fragment_mix_mat;
            yiq2.yz *= vec2(i_mod2, q_mod2);
            yiq2.yz = mix(yiq2.yz, tmp.yz, mit);
            if (res > 1.025) {
                mod_phase2 = chroma_phase2 + pix_no.x * CHROMA_MOD_FREQ * res;
                i_mod2 = cos(mod_phase2);
                q_mod2 = sin(mod_phase2);
                yiqs.yz *= vec2(i_mod2, q_mod2);
                yiq2.x = dot(yiqs, fragment_mix_mat[0]);
            }
        }
        float chroma_phase = phase < 2.5 ? 3.1415927 * (mod(line_idx, 2.) + crawl_field) : 0.6667 * 3.1415927 * (mod(line_idx, 3.) + crawl_field);
        float mod_phase = chroma_phase + pix_no.x * CHROMA_MOD_FREQ;
        float i_mod = cos(mod_phase);
        float q_mod = sin(mod_phase);
        yiq.yz *= vec2(i_mod, q_mod);
        yiq *= fragment_mix_mat;
        yiq.yz *= vec2(i_mod, q_mod);
        yiq.yz = mix(yiq.yz, tmp.yz, mit);
        if (res > 1.025) {
            mod_phase = chroma_phase + pix_no.x * CHROMA_MOD_FREQ * res;
            i_mod = cos(mod_phase);
            q_mod = sin(mod_phase);
            yiqz.yz *= vec2(i_mod, q_mod);
            yiq.x = dot(yiqz, fragment_mix_mat[0]);
        }
        if (params.ntsc_phase == 4.) {
            yiq.x = lum;
            yiq2.x = lum;
        }
        if (MERGE > 0.5) {
            if (params.ntsc_rainbow1 < 0.5 || phase > 2.5) yiq = 0.5 * (yiq + yiq2); else yiq.x = 0.5 * (yiq.x + yiq2.x);
        }
        FragColor = vec4(yiq, lum);
    }
}

vec4 hook() {
    vertex_main();
    fragment_main();
    return FragColor;
}

const int fragment_TAPS_2_phase = 32;
const int fragment_TAPS_3_phase = 24;
const float fragment_luma_filter_2_phase[33] = float[33](-0.000174844, -0.000205844, -0.000149453, -0.000051693, 0., -0.000066171, -0.000245058, -0.000432928, -0.000472644, -0.000252236, 0.000198929, 0.000687058, 0.000944112, 0.000803467, 0.000363199, 0.000013422, 0.000253402, 0.001339461, 0.002932972, 0.003983485, 0.003026683, -0.001102056, -0.008373026, -0.0168977, -0.02291448, -0.021642348, -0.028863274, 0.027271956, 0.05492192, 0.09834258, 0.13904428, 0.16805583, 0.17857143);
const float fragment_chroma_filter_3_phase[33] = float[33](0., 0., 0., 0., 0., 0., 0., 0., -0.000118847, -0.000271306, -0.000502642, -0.000930833, -0.001451013, -0.002064744, -0.002700432, -0.003241276, -0.003524948, -0.003350284, -0.002491729, -0.000721149, 0.002164659, 0.006313635, 0.011789103, 0.01854566, 0.026414396, 0.03510071, 0.04419657, 0.053207204, 0.061590277, 0.0688036, 0.07435619, 0.07785656, 0.079052396);
void fragment_main() {
    if (vTexCoord.x * params.speedup > 1.00325) discard;
    float luma_filter_3_phase[33] = float[33](0., 0., 0., 0., 0., 0., 0., 0., -0.00001202, -0.000022146, -0.000013155, -0.00001202, -0.000049979, -0.00011394, -0.00012215, -0.000005612, 0.000170516, 0.000237199, 0.00016964, 0.000285688, 0.000984574, 0.002018683, 0.002002275, -0.005909882, -0.012049081, -0.01822286, -0.022606932, 0.00246086, 0.035868224, 0.08401645, 0.1355635, 0.17526127, 0.22017655);
    // PATCHED: must track pass1's calibration or the filter taps stop lining up
    // with the modulation (one_x = 0.25 * OriginalSize.z / res).
    float res = params.ntsc_scale * fragment_auto_rez * params.nscale * (ntsc_ref_width / params.OriginalSize.x);
    ;
    float OriginalSize = params.OriginalSize.x * fragment_auto_rez;
    vec2 one_x = 0.25 * params.OriginalSize.zz / res / params.speedup;
    vec3 signal = vec3(0.);
    float phase = params.ntsc_phase < 1.5 ? OriginalSize > 300. ? 2. : 3. : params.ntsc_phase > 2.5 ? 3. : 2.;
    if (params.ntsc_phase > 3.5) {
        phase = 3.;
        luma_filter_3_phase = fragment_luma_filter_2_phase;
    }
    float offset = 0.;
    int i = 0;
    float j = 0.;
    vec3 wsum = 0..xxx;
    vec3 sums = wsum;
    vec3 tmp = wsum;
    if (phase < 2.5) {
        float iloop = max(params.ntsc_taps, 8.);
        if (params.ntsc_charp > 0.25) iloop = min(iloop, 14.);
        int loopstart = int(fragment_TAPS_2_phase - iloop);
        float ltap = iloop + 1.;
        float cs_sub = iloop - iloop / params.ntsc_cscale;
        float ctap = 0.;
        float mit = 1. + 0.04 * pow(fragment_smothstep(16., 8., iloop), 0.5);
        vec2 dx = vec2(one_x.x * mit, 0.);
        vec2 dx1 = dx;
        for (i = loopstart; i < 32; i++) {
            offset = float(i - loopstart);
            j = offset + 1.;
            dx1 = (offset - iloop) * dx;
            sums = fragment_fetch_offset2(dx1);
            ctap = max(j - cs_sub, 0.);
            tmp = vec3(fragment_luma_filter_2_phase[max(i, 0)], ctap.xx);
            wsum = wsum + tmp;
            signal += sums * tmp;
        }
        ctap = ltap - cs_sub;
        tmp = vec3(fragment_luma_filter_2_phase[fragment_TAPS_2_phase], ctap.xx);
        wsum = wsum + wsum + tmp;
        signal += texture(Source, vTexCoord).xyz * tmp;
        signal = signal / wsum;
    } else {
        one_x.y = one_x.y / params.ntsc_cscale1;
        vec3 dx = vec3(one_x.x, one_x.y, 0.);
        float mit = 1.;
        float iloop = min(params.ntsc_taps, fragment_TAPS_3_phase);
        if (params.ntsc_phase > 3.5) {
            iloop = max(iloop, 8.);
            mit = 1. + 0.04 * pow(fragment_smothstep(16., 8., iloop), 0.5);
        }
        vec3 dx1 = dx;
        dx.x *= mit;
        int loopstart = int(32. - iloop);
        for (i = loopstart; i < 32; i++) {
            offset = float(i - loopstart);
            dx1.xy = (offset - iloop) * dx.xy;
            sums = fragment_fetch_offset3(dx1);
            tmp = vec3(luma_filter_3_phase[i], fragment_chroma_filter_3_phase[i].xx);
            wsum = wsum + tmp;
            signal += sums * tmp;
        }
        tmp = vec3(luma_filter_3_phase[fragment_TAPS_2_phase], fragment_chroma_filter_3_phase[fragment_TAPS_2_phase], fragment_chroma_filter_3_phase[fragment_TAPS_2_phase]);
        wsum = wsum + wsum + tmp;
        signal += texture(Source, vTexCoord).xyz * tmp;
        signal = signal / wsum;
    }
    signal.x = clamp(signal.x, 0., 1.);
    if (params.ntsc_ring > 0.05) {
        vec2 dx = vec2(params.OriginalSize.z / min(res, 1.) / params.speedup, 0.);
        float a = texture(Source, vTexCoord - 2. * dx).a;
        float b = texture(Source, vTexCoord - dx).a;
        float c = texture(Source, vTexCoord + 2. * dx).a;
        float d = texture(Source, vTexCoord + dx).a;
        float e = texture(Source, vTexCoord).a;
        signal.x = mix(signal.x, clamp(signal.x, min(min(min(a, b), min(c, d)), e), max(max(max(a, b), max(c, d)), e)), params.ntsc_ring);
    }
    float orig = fragment_get_luma(texture(PrePass0, vTexCoord * vec2(params.speedup, 1.)).rgb);
    FragColor = vec4(signal, orig);
}

    vertex_main();
    fragment_main();
    return FragColor;

vec2 fragment_psrnd(vec2 TexCoord, float FrameCount) {
    vec3 p = vec3(TexCoord * 16758.545, FrameCount);

vec3 fragment_Noise(vec2 TexCoord, float FrameCount) {
    TexCoord = fragment_psrnd(TexCoord, FrameCount);
    vec3 p = vec3(TexCoord * 758.5453, FrameCount);
    float res1 = fract(sin(dot(p, vec3(12.989835, 78.23341, 0.16453))) * 43758.547);
    float res2 = fract(sin(dot(p, vec3(39.34679, 11.13523, 83.15573))) * 39459.324);
    float res3 = fract(sin(dot(p, vec3(73.15691, 52.23504, 9.15197))) * 60493.848);
    float lns = float(abs(res1 - 0.5) < 0.2 * fragment_RFNOISE);
    float cny = float(abs(res2 - 0.5) < 0.5 * fragment_RFNOISE);
    float cnz = float(abs(res3 - 0.5) < 0.5 * fragment_RFNOISE);
    lns = lns * fragment_smothstep(0.5 - 0.2 * fragment_RFNOISE, 0.5 + 0.2 * fragment_RFNOISE, res1) * 0.66 - 0.33 * lns;
    cny = cny * fragment_smothstep(0.5 - 0.5 * fragment_RFNOISE, 0.5 + 0.5 * fragment_RFNOISE, res2) - 0.5 * cny;
    cnz = cnz * fragment_smothstep(0.5 - 0.5 * fragment_RFNOISE, 0.5 + 0.5 * fragment_RFNOISE, res3) - 0.5 * cnz;
    return vec3(lns, cny, cnz);
}

// PATCHED: the rainbow pass has its own copy of the temporal state machine.
// Its line-shift term alternates the sampled neighbour row with field parity,
// which is only meaningful while the chroma phase itself is advancing. Same
// resolution as pass 1 - see fragment_resolve_crawl_states() there. Both passes
// see the same params.OriginalSize (MAIN_RGB), so they agree without talking.
float fragment_resolve_line_period() {
    if (ntsc_line_period > 0.5) return ntsc_line_period;
    float lines = params.OriginalSize.y;
    bool sd = lines > 288.5 && lines < 576.5;
    bool hd = lines > 1079.5 && lines < 1080.5;
    return sd || hd ? 2. : 1.;
}

float fragment_resolve_crawl_states(float line_period, float phase) {
    if (ntsc_crawl_states > 0.5) return ntsc_crawl_states;
    if (line_period > 1.5) return 4.;
    return phase > 2.5 ? 3. : 1.;
}

void fragment_main() {
    if (params.speedup > 1.25 && (params.OriginalSize.y > 500. && params.SourceSize.y > 820. || params.SourceSize.y > 820.)) {
        vec3 signal = texture(Source, vTexCoord0).xyz;
        signal.x = pow(signal.x, 1. / params.ntsc_gamma);
        FragColor = vec4(clamp(fragment_yiq2rgb(signal), 0., 1.), 1.);
    } else {
        vec2 xx = vec2(0.5 * params.OriginalSize.z / fragment_auto_rez / params.speedup, 0.);
        vec2 dx = vec2(0.0625 * params.OriginalSize.z / fragment_auto_rez / params.speedup, 0.);
        vec2 texcoord0 = (floor(params.OriginalSize.xy * vTexCoord) + 0.5) * params.OriginalSize.zw;
        vec2 texcoord = vTexCoord - 2. * dx;
        vec2 vTex = vTexCoord;
        if (params.speedup > 1.25) {
            texcoord0 = vTexCoord1;
            texcoord = vTexCoord0;
            vTex = vTexCoord0;
        }
        vec3 l1 = texture(Source, texcoord + xx).xyz;
        vec3 l2 = texture(Source, texcoord - xx).xyz;
        float dy = 0.;
        vec2 yy = vec2(0., params.OriginalSize.w);
        xx = vec2(params.OriginalSize.z / fragment_auto_rez / params.speedup, 0.);
        float phase = params.ntsc_phase < 1.5 ? params.OriginalSize.x * fragment_auto_rez > 300. ? 2. : 3. : params.ntsc_phase > 2.5 ? 3. : 2.;
        if (params.ntsc_phase > 3.5) phase = 3.;
        float th = phase < 2.5 ? 0.025 : 0.0075;
        float ca = texture(NPass1, texcoord0 - xx - xx).a;
        float c0 = texture(NPass1, texcoord0 - xx).a;
        float c1 = texture(NPass1, texcoord0).a;
        float c2 = texture(NPass1, texcoord0 + xx).a;
        float cb = texture(NPass1, texcoord0 + xx + xx).a;
        float line0 = fragment_smothstep(th, 0., min(abs(c1 - c0), abs(c2 - c1)));
        float line1 = max(fragment_smothstep(th, 0., min(abs(ca - c0), abs(c2 - cb))), line0);
        float line2 = max(fragment_smothstep(th, 0., min(abs(ca - c2), abs(c0 - cb))), line1);
        float diffboost = 0.;
        if (params.ntsc_fonts > 0.255) {
            float b1 = texture(NPass1, texcoord0 - yy).a;
            float d1 = texture(NPass1, texcoord0 + yy).a;
            float d2 = texture(NPass1, texcoord0 + xx + yy).a;
            float maxdif = float(0.5 * (abs(c0 - c1) + abs(c1 - c2)) > 1. - params.ntsc_fonts);
            float linecb = fragment_smothstep(0.1, 0., max(abs(c1 - b1), abs(c1 - d1)));
            linecb = linecb * float(abs(c1 - b1) < 0.05 || abs(c2 - d2) < 0.05);
            float xdif1 = 0.;
            float xdif2 = 0.;
            float th = 0.215;
            for (float i = 0.5; i < 10.5; i++) {
                float j = i + 1.;
                ca = texture(Source, texcoord - i * xx).x;
                c0 = texture(Source, texcoord - j * xx).x;
                c2 = texture(Source, texcoord + i * xx).x;
                cb = texture(Source, texcoord + j * xx).x;
                xdif1 = max(abs(ca - c0) - th, xdif1);
                xdif2 = max(abs(cb - c2) - th, xdif2);
            }
            float linex = fragment_smothstep(0., 0.125, min(xdif1, xdif2));
            diffboost = min(linex * linecb * maxdif, 0.625);
        }
        vec3 ref = texture(Source, texcoord).xyz;
        vec2 orig = ref.yz;
        if (params.ntsc_rainbow1 > 0.5 && (phase < 2.5 || params.ntsc_phase == 5.)) {
            float ybool = 1.;
            if (params.ntsc_rainbow1 < 1.5 && bool(line0)) ybool = 0.; else if (params.ntsc_rainbow1 < 2.5 && bool(line2)) ybool = 0.;
            float line_no = floor(mod(params.OriginalSize.y * vTexCoord.y, 2.));
            // PATCHED: frozen crawl means a frozen comb direction. This term
            // flips which neighbour row feeds the rainbow every field, standing
            // in for the field-dependent chroma comb. When the chroma phase is
            // not advancing (progressive 2-phase, crawl_states == 1) there is no
            // field parity to track, and leaving it running adds a 30 Hz flicker
            // on top of a pattern that should be nailed to the picture.
            float rainbow_states = fragment_resolve_crawl_states(fragment_resolve_line_period(), phase);
            float frame_no = rainbow_states < 1.5 ? 0.
                : floor(mod(ntsc_frame_hold >= 0. ? ntsc_frame_hold
                            : floor(float(params.FrameCount) / ntsc_frame_div), 2.));
            float ii = abs(line_no - frame_no);
            dy = ii * params.OriginalSize.w * ybool;
            vec2 ref1 = texture(Source, texcoord - vec2(0., dy)).yz;
            vec2 ref2 = texture(Source, texcoord + vec2(0., dy)).yz;
            vec2 rdf1 = abs(orig - ref1);
            vec2 rdf2 = abs(orig - ref2);
            rdf1 = rdf1 / max(rdf1 + rdf2, 0.0000001);
            ref.yz = mix(ref1, ref2, rdf1);
        }
        float lum1 = min(texture(NPass1, vTex - dx).a, texture(NPass1, vTex + dx).a);
        float lum2 = ref.x;
        vec3 l3 = abs(l1 - l2);
        float dif = max(max(max(l3.x, l3.y), max(l3.z, abs(l1.x * l1.x - l2.x * l2.x))), diffboost);
        float dff = pow(dif, 0.125);
        float lc = fragment_smothstep(0.2, 0.1, abs(lum2 - lum1)) * dff;
        float tmp = fragment_smothstep(0.05 - 0.03 * lc, 0.425 - 0.375 * lc, dif);
        float tmp1 = pow((tmp + 0.1) / 1.1, 0.25);
        float sweight = mix(tmp, tmp1, line0);
        float sweighr = mix(tmp, tmp1, line2);
        vec3 signal = ref;
        float ntsc_sharp = abs(params.ntsc_sharp);
        if (ntsc_sharp > 0.25) {
            float mixer = sweight;
            if (params.ntsc_sharp > 0.25) mixer = sweighr;
            mixer *= 0.1 * ntsc_sharp;
            float lummix = mix(lum2, lum1, mixer);
            float lm1 = mix(lum2 * lum2, lum1 * lum1, mixer);
            lm1 = sqrt(lm1);
            float lm2 = mix(sqrt(lum2), sqrt(lum1), mixer);
            lm2 = lm2 * lm2;
            float k1 = abs(lummix - lm1) + 0.00001;
            float k2 = abs(lummix - lm2) + 0.00001;
            signal.x = min((k2 * lm1 + k1 * lm2) / (k1 + k2), 1.);
            signal.x = min(signal.x, max(params.ntsc_shape * signal.x, lum2));
        }
        if (params.ntsc_charp + params.ntsc_charp3 > 0.25) {
            dx.x = 0.0625 * params.OriginalSize.z / fragment_auto_rez;
            float texcoordx = params.OriginalSize.x * (vTexCoord.x + dx.x) - 0.5;
            float fpx = fract(texcoordx);
            texcoordx = (floor(texcoordx) + 0.5) * params.OriginalSize.z;
            float mixer = sweight;
            if (params.ntsc_sharp > 0.25) mixer = sweighr;
            mixer = mix(fragment_smothstep(0.075, 0.125, max(l3.y, l3.z)), fragment_smothstep(0.015, 0.0275, dif), line2) * mixer;
            mixer *= 0.1 * (phase < 2.5 ? params.ntsc_charp : params.ntsc_charp3);
            texcoord = vec2(texcoordx, texcoord.y);
            vec3 orig_ch = fragment_rgb2yiq(mix(texture(PrePass0, texcoord).rgb, texture(PrePass0, texcoord + 16. * dx).rgb, clamp(1.5 * fpx - 0.25, 0., 1.)));
            signal.yz = mix(signal.yz, orig_ch.yz, mixer);
        }
        if (params.RFNOISE1 + params.RFNOISE2 > 0.005) {
            vec3 ns = fragment_Noise(vTexCoord, float(params.FrameCount));
            float Y = mix(0.375, 1., pow(signal.x, 0.2));
            signal.x = clamp(signal.x + params.RFNOISE1 * ns.x, 0., 1.);
            signal.y = clamp(signal.y + Y * params.RFNOISE2 * (0.325 + 1.325 * abs(signal.y)) * ns.y, -0.6, 0.6);
            signal.z = clamp(signal.z + Y * params.RFNOISE2 * (0.325 + 1.325 * abs(signal.z)) * ns.z, -0.53, 0.53);
        }
        signal.x = pow(signal.x, 1. / params.ntsc_gamma);
        signal = clamp(fragment_yiq2rgb(signal), 0., 1.);
        sweighr = phase == 2. || params.ntsc_phase > 3.5 ? sweighr : 1.;
        FragColor = vec4(signal, sweighr);
    }
}

vec4 hook() {
    vertex_main();
    fragment_main();
    vec4 c = delinearize(FragColor);
    // This pass replaces MAIN. ntsc-pass3 writes a scanline weight into alpha
    // (harmless in RetroArch, which discards final-pass alpha), but mpv
    // composites it - and --alpha=blend-tiles draws transparency as a
    // white/grey checkerboard. Earlier passes DO carry signal data in alpha
    // (pass3 reads NPass1.a), so only this final pass may be forced opaque.
    c.a = 1.0;
    return c;
}
