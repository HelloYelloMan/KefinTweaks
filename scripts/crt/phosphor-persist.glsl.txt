// ---------------------------------------------------------------------------
// phosphor-persist.glsl -- CRT phosphor persistence as a standalone post-pass
// ---------------------------------------------------------------------------
// Nothing else in the chain models persistence. crt-royale is a purely spatial
// shader: every "phosphor" in it is mask geometry (triad tiles), not decay. So
// each rendered field is drawn from scratch and the previous field's light is
// gone the instant it is replaced. A real CRT does not do that, and the visible
// consequence in this chain is interline twitter -- crt-royale bob-fields a
// 480-line frame at 60 Hz and the rows the beam did not light this field go to
// an interpolated average instead of staying lit.
//
// This file keeps a copy of what the screen was emitting last frame and lets it
// decay into this one.
//
// WHY THIS IS A SEPARATE FILE AND NOT A PASS INSIDE CRT-ROYALE
//
// Because the two stages it sits behind are time-invariant, so they commute
// with the decay. For a per-pixel mask m:
//
//     max(m*c, w * m*p)  ==  m * max(c, w*p)
//
// and the geometry warp is a static remap, so holding history in warped screen
// space is the same as warping a decayed image. Running the decay here is
// therefore EXACTLY equivalent to running it before the mask and the warp.
//
// What it is not equivalent to is running it before the bloom path. Halation,
// diffusion and BLOOM_APPROX are spatial blurs of the CURRENT field, and blur
// does not commute with the decay, so the afterglow does not scatter in the
// faceplate the way it should. That is second-order, and buying it means adding
// a pass inside crt-royale-super.glsl, which means updating
// `const float royale_pass_count = 15.;` at all fifteen sites -- miss one and
// royale_true_frame() silently desyncs, taking field parity, the interlacing
// and the NTSC dot-crawl alignment with it. Not worth it until the effect has
// earned its place.
//
// THE MODEL
//
//     shown   = max(current, history * weights)
//     history = max(current * initial, history * weights)
//
// max(), not mix(). A phosphor struck by the beam jumps straight to full output
// -- it does not average with its own afterglow. That distinction is the whole
// point on interline twitter: with mix() a freshly lit row comes up dimmed by
// whatever was there before, which is the opposite of what a CRT does.
//
// The two lines differ only in `initial`, and that split is what makes an
// initial-opacity control possible at all. The naive way to dim a trail is to
// scale the afterglow on the way out -- but that scale then lands in the stored
// value too, so it re-applies every tick and turns into a second decay term
// (w*a per step instead of w). The trail gets SHORTER, not dimmer, which is not
// what the knob says. Attenuating only where fresh light ENTERS the history
// buffer applies the drop exactly once, at handoff:
//
//     tick 0: shown = C            history = C*a
//     tick 1: shown = C*a*w        history = C*a*w
//     tick n: shown = C*a*w^n
//
// so `initial` sets where the trail starts and `decay` sets how fast it falls,
// and the two knobs are orthogonal. Setting initial = 1 recovers the original
// single-line model exactly.
//
// Two useful properties fall out of max-and-multiply:
//   - It is idempotent under redraw. mpv re-runs the whole chain when it
//     redraws a paused frame, and `frame` cannot tell that from a new video
//     frame (see the libplacebo pass-counter note in the crt-royale header).
//     Here it does not matter: repeated application converges to `current`
//     rather than drifting, so pausing settles instead of fading.
//   - It commutes with a pure power law, since (w*p)^g == w^g * p^g. So the
//     choice of working space only reparametrises the weights, it does not
//     distort the curve.
//
// PHYSICAL DECAY MODEL, AND WHY IT IS PARAMETRISED RATHER THAN OBEYED
//
// P22 primary decay is fast: roughly 1.5 ms for green and blue and around 6 ms
// for red, with 10%-of-peak figures for the whole screen landing in the 1.5-6.4
// ms band depending on who is measuring and how. Every one of those numbers is
// SHORTER than a single 60 Hz video frame, and most of them are comparable to
// one 240 Hz refresh tick. Run the honest numbers: tau = 1.5 ms against a
// 4.167 ms tick gives exp(-2.78) = 0.06, so the afterglow is down 94% before
// the panel has drawn it once. It is not that a 240 Hz display is too slow to
// show phosphor decay -- it is that at physical time constants there is
// essentially nothing left to show by the first tick, on any display.
//
// What people actually SEE on a CRT and call persistence is mostly the tail,
// not the primary decay. P22 is not a single exponential: the initial fall is
// roughly exponential and is then followed by a much slower power-law tail
// (I = I0*(1+a*t)^-n), and that low-amplitude tail is what carries a visible
// smear across many frames. This shader models only the exponential term, so
// any setting long enough to see is already outside the physics it implements.
//
// WHAT persist_decay MEANS, AND WHY IT IS NORMALISED ON GREEN
//
// It is GREEN's tau, in milliseconds, and every other channel is a fixed ratio
// to it: red 4x, blue 1.2x. At the default 1.5 that is R 6.0, G 1.5, B 1.8 --
// the published pair above, readable straight off the knob. persist_phosphor
// fades those ratios toward 1, so at 0 all three channels decay at exactly
// persist_decay and at 1 they take the spread in full. Green holds still either
// way, which is the point: the knob means one thing regardless of the spread.
//
// This was previously normalised so that dot(tau, luma) == persist_decay -- a
// luma-weighted mean, under which NO channel ever equalled the knob (red was
// 2.42x it, green 0.605x) and its meaning slid as persist_phosphor moved. That
// form had one genuine advantage, worth recording because it was given up
// deliberately: it held PERCEIVED afterglow constant across the spread sweep,
// so turning persist_phosphor up redistributed decay between channels without
// changing how much the picture glowed overall. Normalising on green does not
// -- raising the spread strictly ADDS afterglow, because red stretches 4x and
// nothing gets shorter. That was judged the lesser cost: a value quoted in
// milliseconds should be checkable against a datasheet, and it protects a knob
// set once. The visible consequence is that a given persist_decay now runs
// about 1.65x longer than it did before the change.
//
// Hence `persist_initial`. Treat this file as a look control with physically
// derived UNITS rather than a simulation: `persist_decay` in real milliseconds
// picks the shape of the falloff, `persist_initial` decides how much of it you
// are actually allowed to see. Dialing decay up to fake a visible tail also
// stretches the bright head of the trail into obvious smear; dropping initial
// instead keeps the head where the beam put it and lets the tail sit under the
// image the way a real faceplate's does.
//
// THE TICK IS NOT SELF-DISCOVERABLE
//
// tau only means milliseconds if the shader knows how long a tick is, and
// libplacebo hands a user shader no clock at any stage -- frame, random,
// input_size, target_size, tex_offset, and nothing else. This was checked
// rather than assumed. `frame` is a perfectly good tick COUNTER
// here -- this is a single-pass file, so it advances exactly once per presented
// frame with no division needed -- but counting is not a clock, and it cannot
// say how many milliseconds a tick is. persist_tick_ms is that missing constant,
// supplied from outside, and it is the ONLY thing the parameter does: it never
// affects when this hook runs, only how much decay one run applies.
//
// The value is 1000 / display-fps. retro-base sets interpolation=yes and
// video-sync=display-resample for every mode, so OUTPUT runs once per REFRESH,
// not once per video frame, and all four mode profiles pass 4.167 for a 240 Hz
// panel. It cannot live in retro-base: the mode profiles set glsl-shader-opts
// wholesale and would wipe an append there -- see the comment at that profile.
//
// Only the RATIO persist_tick_ms / persist_decay reaches the GPU, so getting it
// wrong desyncs nothing, it just rescales the decay. Declaring 16.667 while the
// real interval is 4.167 makes each tick decay as much as 16.667 ms should while
// ticks keep arriving 240 times a second: the trail dies 4x too fast in wall
// clock, which reads as the effect being absent rather than wrong.
//
// PLACEMENT: last in the chain, after crt-royale.
//
//   glsl-shaders=...:crt-royale-super.glsl:phosphor-persist.glsl
//
// THE STORAGE TEXTURE
//
// TWO PARSER RULES, BOTH LEARNED THE HARD WAY ON THIS FILE:
//
// 1. Do not write a directive marker (two slashes then a bang) inside a
//    comment. libplacebo finds block boundaries by searching for that marker
//    ANYWHERE in the file, not just at the start of a line, so a directive
//    named in prose is parsed as a directive and the whole shader fails with
//    "Unrecognized command". Directive names below are written bare.
// 2. Do not put a comment between a PARAM block's value and the next
//    directive. A block's payload runs to the next marker, so a comment there
//    becomes part of the value: "Failed parsing variable data: 0.2".
//    Comments are only safe before the FIRST directive in the file (i.e. up
//    here) or inside a hook's GLSL body.
//
// The format is rgba16hf, NOT rgba16f. Both names exist in libplacebo's
// Vulkan format table, but rgba16f is the EMUFMT entry -- the emulated variant
// carrying fp32 on the host and fp16 on the GPU, meant for data upload. The
// native storable one is REGFMT rgba16hf.
//
// libplacebo's STORAGE directive binds a texture as a readwrite coherent
// image2D instead of a read-only sampler2D, and unlike a SAVE target it is not
// recycled between frames. Normally a TEXTURE block must carry a hex payload
// matching its SIZE exactly, which for 3840x2160 rgba16hf would be ~63 MB of
// hex; storable textures are the documented exception ("it's okay that the
// texture has no initial data" in libplacebo's parser), so this one is empty.
//
// SIZE takes literal integers -- no size expressions -- so it cannot track
// the window. It is allocated at panel size and only the top-left sub-rect is
// used. A window LARGER than this degrades safely rather than corrupting:
// out-of-range imageStore is discarded and out-of-range imageLoad returns zero,
// so the excess simply gets no persistence. Bump the numbers if the panel does.
//
// Each invocation touches only its own texel, so the coherent image needs no
// synchronisation -- there is no cross-invocation hazard to worry about.
// ---------------------------------------------------------------------------

//!PARAM persist_initial
//!DESC Phosphor Persistence - afterglow opacity at handoff (0 = off, pass skipped)
//!TYPE DYNAMIC float
//!MINIMUM 0
//!MAXIMUM 1
0

//!PARAM persist_decay
//!DESC Phosphor Persistence - GREEN's decay time constant in ms (red is 4x, blue 1.2x)
//!TYPE DYNAMIC float
//!MINIMUM 0.1
//!MAXIMUM 20.0
1.5

//!PARAM persist_tick_ms
//!DESC Phosphor Persistence - milliseconds per OUTPUT run (16.667 = 60Hz, 4.167 = 240Hz)
//!TYPE DYNAMIC float
//!MINIMUM 1.0
//!MAXIMUM 40.0
4.167

//!PARAM persist_phosphor
//!DESC Phosphor Persistence - P22 spread, 0 = all channels at persist_decay, 1 = full R4x B1.2x
//!TYPE DYNAMIC float
//!MINIMUM 0
//!MAXIMUM 1
1

//!PARAM persist_gamma
//!DESC Phosphor Persistence - power-law exponent, used when persist_trc = 0
//!TYPE DYNAMIC float
//!MINIMUM 1
//!MAXIMUM 3
2.2

//!PARAM persist_trc
//!DESC Phosphor Persistence - transfer function of the OUTPUT buffer (0 power law, 1 PQ, 2 HLG)
//!TYPE DYNAMIC int
//!MINIMUM 0
//!MAXIMUM 2
0

//!TEXTURE PHOSPHOR
//!SIZE 3840 2160
//!FORMAT rgba16hf
//!STORAGE

//!HOOK OUTPUT
//!BIND HOOKED
//!BIND PHOSPHOR
//!DESC CRT phosphor persistence
//!WHEN persist_initial 0.0 >

// Working space. The decay is a MULTIPLY, so it has to happen in light-linear
// values or the curve is wrong -- and this hook runs after mpv's output colour
// transform, so the encoding here is the TARGET's, not the source's. Those are
// the same thing only because mpv.conf turns on target-colorspace-hint with
// hint-mode=source for PQ/HLG in fullscreen; the Lua that drives persist_trc
// mirrors that same condition. Everything else lands on SDR, where the power
// law at persist_gamma is right.
//
// linearize()/delinearize() are deliberately not used: those helpers are well
// defined at MAIN and relying on them at OUTPUT would be an unverified
// assumption, so the encode/decode below is explicit instead.
//
// max() itself does not care -- it commutes with any monotonic invertible
// encoding -- so getting this wrong distorts the decay curve without breaking
// the model.

const float PQ_M1 = 0.1593017578125, PQ_M2 = 78.84375;
const float PQ_C1 = 0.8359375, PQ_C2 = 18.8515625, PQ_C3 = 18.6875;
const float HLG_A = 0.17883277, HLG_B = 0.28466892, HLG_C = 0.55991073;

vec3 persist_decode(vec3 v) {
    v = max(v, 0.0);
    if (persist_trc == 1) {
        vec3 e = pow(v, vec3(1.0 / PQ_M2));
        return pow(max(e - PQ_C1, 0.0) / max(PQ_C2 - PQ_C3 * e, 1e-6), vec3(1.0 / PQ_M1));
    }
    if (persist_trc == 2) {
        return mix((exp((v - HLG_C) / HLG_A) + HLG_B) / 12.0,
                   v * v / 3.0,
                   step(v, vec3(0.5)));
    }
    return pow(v, vec3(persist_gamma));
}

vec3 persist_encode(vec3 l) {
    l = max(l, 0.0);
    if (persist_trc == 1) {
        vec3 e = pow(l, vec3(PQ_M1));
        return pow((PQ_C1 + PQ_C2 * e) / (1.0 + PQ_C3 * e), vec3(PQ_M2));
    }
    if (persist_trc == 2) {
        return mix(HLG_A * log(max(12.0 * l - HLG_B, 1e-6)) + HLG_C,
                   sqrt(max(3.0 * l, 0.0)),
                   step(l, vec3(1.0 / 12.0)));
    }
    return pow(l, vec3(1.0 / persist_gamma));
}

vec3 p22_refresh_weights(float decay_ms, float spread) {
    if (decay_ms <= 0.0) return vec3(0.0);

    // How much simulated time one run of this hook represents. Not derivable
    // in-shader; see the tick note in the header.
    float dt = max(persist_tick_ms, 1e-3);

    // P22 per-channel relative decay, normalised ON GREEN so persist_decay IS
    // green's tau in milliseconds and red is exactly 4x it. Green is the
    // reference because that is the channel the published figures quote.
    const vec3 p22 = vec3(4.0, 1.0, 1.2);
    vec3 ratio = mix(vec3(1.0), p22 / p22.g, spread);

    vec3 tau = max(vec3(decay_ms) * ratio, vec3(1e-4));

    // Exact exponential decay factor over one tick of persist_tick_ms
    return exp(-dt / tau);
}

vec4 hook() {
    ivec2 texel = ivec2(gl_FragCoord.xy);
    vec4 c = HOOKED_tex(HOOKED_pos);

    // Decode from OUTPUT space into light-linear
    vec3 current = persist_decode(c.rgb);

    vec3 weights = p22_refresh_weights(persist_decay, persist_phosphor);

    // max() guards the negative values mpv's linearize() can emit near black,
    // which would otherwise be latched into the history buffer and persist.
    vec3 afterglow = max(imageLoad(PHOSPHOR, texel).rgb, 0.0) * weights;

    // What the screen emits: the beam wins outright wherever it struck.
    vec3 lit = max(current, afterglow);

    // What the buffer carries forward: fresh light is attenuated ONCE as it
    // enters the history, so persist_initial sets the trail's starting level
    // without also compounding into its decay rate. See the header.
    imageStore(PHOSPHOR, texel, vec4(max(current * persist_initial, afterglow), 1.0));

    // Encode back into OUTPUT space. Clamp after, not before: `lit` never
    // exceeds `current` where the beam is lit, so the only thing this bounds is
    // encoder overshoot near the top of the range.
    return vec4(clamp(persist_encode(lit), 0.0, 1.0), c.a);
}
