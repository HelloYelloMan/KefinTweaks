# The retro CRT pipeline, as named profiles that are NOT applied at startup.
#
# mpv.conf includes this file so the profiles get *defined*; nothing here takes
# effect until something runs `apply-profile`. scripts/retro-toggle.lua
# handles cycling through these profiles, bound to CTRL+r in input.conf.
#
# profile-restore=copy makes the toggles reversible: applying a profile
# snapshots the previous value of every option it touches (vf, glsl-shaders,
# glsl-shader-opts), and `apply-profile <name> restore` puts those values back.

[retro-base]
profile-restore=copy
# The three vf filters below are CPU filters, and only the -copy hwdec
# variants bring decoded frames back to system RAM where a CPU filter can
# reach them. mpv's own config uses nvdec-copy globally so this is a no-op
# there; the shim runs plain nvdec, and without this its retro chain would
# have no frames to filter. profile-restore puts its value back on the way out.
hwdec=nvdec-copy
vf=sub
vf-add=fps=60000/1001
vf-add=@autocrop:crop=out_w=iw:out_h=ih:x=0:y=0
dither=fruit

interpolation=yes
video-sync=display-resample
tscale=nearest
# DISPLAY-RATE RENDERING. phosphor-persist.glsl hooks OUTPUT, which libplacebo
# applies after the interpolation blend, once per PRESENTED frame. Everything
# else in the chain hooks MAIN and is cached per source frame. These three
# options are what put OUTPUT on the refresh clock:
#
#   interpolation=yes             Render per refresh instead of per video frame.
#                                 INERT on its own -- mpv ignores it unless
#                                 video-sync is a display-* mode, and only logs
#                                 a warning rather than failing.
#
#   video-sync=display-resample   What actually presents at refresh rate. It
#                                 nudges playback speed so 59.94 fps content
#                                 lands on precisely four refreshes per field at
#                                 240 Hz rather than 4.004.
#
#   tscale=nearest                NO cross-frame blending, at all. This is the
#                                 one that has to be exact, and "narrow" was not
#                                 good enough -- see below. Do NOT put a smooth
#                                 tscale here, and do not put oversample back.
#
# WHY NOT oversample (it was here until 2026-08-05, and it was the bug)
#
# crt-royale BOB-FIELDS. Adjacent rendered frames do not carry near-identical
# pictures the way progressive video does -- they carry DIFFERENT INTERLACE
# FIELDS, with the lit scanlines in alternating rows. So any blend of two
# consecutive frames merges the two fields into a half-brightness full raster.
# Frame mixing is the one thing interpolation adds that touches the picture, and
# on this chain it is not a smoothing operation, it is a structural one.
#
# oversample was chosen as "nearest-neighbour with a narrow blend window", which
# is true and was not sufficient. The window is only zero when the frame
# boundaries land exactly on refreshes. video-sync=display-resample gets the
# ratio to about 4.0002 rather than 4.0000 (measured: source 59.940, display
# 239.751), so the boundary drifts through the refresh grid and a partial blend
# lands periodically. nearest has no window to drift into.
#
# Symptom this fixes: colour crushing under [retro-kurozumi-super] that appeared
# while PLAYING and vanished while PAUSED -- paused, there is no second frame to
# mix, so the blend cannot happen. Confirmed by Robert that interpolation=no made
# it go away; interpolation is kept because it is what puts OUTPUT (and so
# phosphor-persist) on the 240 Hz refresh clock, and nearest removes the blend
# without giving that up.
#
# dither=fruit above is NOT a preference either. Dither runs once per OUTPUT
# frame, so interpolation asks for it 240 times a second instead of 60. The
# error-diffusion setting in mpv.conf has a serial dependency across the whole
# image and at 3840x2160 that saturates an RTX 5080 by itself -- it was first
# misread as the GPU being unable to drive 240 Hz at all. fruit is a cheap
# lookup and is fine on a 10-bit panel. profile-restore=copy snapshots it, so
# toggling the profile off puts mpv.conf's error-diffusion back.
#
# persist_tick_ms does NOT live here, and the reason is a trap worth naming.
# retro-toggle.lua applies retro-base first and the mode profile second, and
# every mode profile below sets `glsl-shader-opts=` WHOLESALE -- so anything
# appended here is silently wiped a moment later. Neither mode profile sets it
# either: it rides on phosphor-persist.glsl's own default of 4.167, which is
# already the 240 Hz value. Set it explicitly in BOTH mode profiles if this ever
# has to change, never here. interpolation=yes above is what makes 4.167 the
# right value: without it OUTPUT runs once per video frame, not once per
# refresh, and the correct figure would be 16.667.

# ---------------------------------------------------------------------------
# crt-royale-super.glsl
# ---------------------------------------------------------------------------
# One shader that is both of the two earlier standalone conversions,
# crt-royale-final.glsl and crt-royale-kurozumi-final.glsl. Two switches select
# the pass topology; everything else was already a shared parameter. See the
# header of crt-royale-super.glsl for the derivation.
#
# Those two files were DELETED on 2026-09-01: super is a strict superset of
# both -- every one of their parameters, plus ten more -- and nothing had
# loaded them since super landed. They are in git history if ever needed.
#
#   royale_grade        0 = no grade pass (crt-royale)   1 = grade (kurozumi)
#   royale_blur_taps    5 = blur5fast (kurozumi)   9 = blur9fast (crt-royale)
#
# There was a third, royale_mpv_bracket. It is gone: the mpv linearize/
# delinearize bracket must be ON whenever grade is off and OFF whenever grade is
# on, so it is now derived from royale_grade inside the shader. Passing it here
# is now an unknown key: no loaded shader declares it, so it does nothing.
#
# The two profiles below were verified against the originals parameter by
# parameter, while those still existed: with these opts applied, every one of
# the 52 params shared with crt-royale-final, and all 105 shared with
# crt-royale-kurozumi-final, held an identical value. The 53 grade-only params are referenced by no pass other than
# grade, so they are inert whenever royale_grade=0.
#
# ---------------------------------------------------------------------------
# THE TWO CHAINS DO NOT CARRY THE SAME PARAMETER SET
# ---------------------------------------------------------------------------
# [retro-super] loads ntsc-adaptive-final.glsl; [retro-kurozumi-super] does not.
# So every ntsc_* / cust_* key, and `quality`, is meaningful in the first and
# undeclared in the second. An undeclared key is inert -- it is not an error and
# nothing else is disturbed -- but it is also not doing what its name suggests,
# which is its own kind of trap: [retro-kurozumi-super] carried quality=0 from
# its introduction until 2026-08-05 and it had never once taken effect.
#
# Before adding a key to either opts line, check it against that profile's own
# glsl-shaders list:
#
#   grep -l '^//!PARAM <key>$' ~/.config/mpv/shaders/retro/*.glsl

# ---------------------------------------------------------------------------
# POSITION-DEPENDENT CONVERGENCE  (stock since 2026-08-05)
# ---------------------------------------------------------------------------
# Declared by crt-royale-super.glsl, so every profile below has it. All four
# are OFF at their defaults and extend the stock convergence_offset_* knobs
# rather than replacing them: those are the order-0 (constant) term of the
# misconvergence field, these are the complete order-2 set, and they simply
# add. Derivation and the parity constraint are in the shader header.
#
#   convergence_h_sides  -8..8    The six convergence_offset_* knobs shift a
#   convergence_h_topbot -8..8    channel uniformly, which no tube does. A real
#   convergence_v_corner -4..4    set is converged at the CENTRE and drifts
#                                 away from it (spec is about 0.1-0.2 mm centre
#                                 against 0.3-0.5 mm corner) because the three
#                                 beams take different-length paths through the
#                                 deflection field.
#
#                                 THE SHAPE IS NOT A FREE CHOICE. Red and blue
#                                 sit either side of green, so they are a
#                                 MIRROR PAIR about the tube's vertical axis,
#                                 and the yoke is mirror-symmetric about it
#                                 too. That forces the horizontal error to be
#                                 EVEN in both axes and the vertical error ODD
#                                 in both. Subtracting the static term, which
#                                 is already convergence_offset_*, exactly
#                                 three second-order terms survive -- and these
#                                 are they. A RADIAL error (r^2, red splaying
#                                 outward) has the wrong parity on both axes;
#                                 it is real, but it is the yoke-tilt and
#                                 neck-magnet MISADJUSTMENT class, not the
#                                 residual a self-converging yoke leaves.
#
#                                 h_sides   horizontal error at the LEFT AND
#                                           RIGHT edges, SAME sense at both
#                                           (service nomenclature: XH).
#                                 h_topbot  horizontal error at the TOP AND
#                                           BOTTOM of a centre VERTICAL line,
#                                           same sense at both (YH). One of the
#                                           residuals a self-converging yoke is
#                                           trimmed against with wedges.
#                                 v_corner  vertical error, which the odd
#                                           parity pins to exactly ZERO along
#                                           both centre lines and largest in
#                                           the CORNERS, flipping sign between
#                                           adjacent ones. That bowtie is the
#                                           classic in-line corner residual.
#
#                                 Each value IS the error at its own
#                                 measurement point -- the edge for the two
#                                 horizontal terms, the corner for the vertical
#                                 one -- so no calibration factor is involved.
#                                 Horizontal is in texels of the 704-wide
#                                 post-NTSC grid, vertical in scanlines out of
#                                 480.
#
#                                 THE AXES NEED VERY DIFFERENT VALUES. Composite
#                                 NTSC has already smeared the horizontal axis
#                                 to about 330 TVL while leaving the vertical
#                                 grid sharp, so horizontal fringing has almost
#                                 nothing to fringe against. The vertical term
#                                 reads at 0.4 where the horizontal ones need
#                                 1-2. That asymmetry is real -- vertical
#                                 misconvergence is the one you notice on a real
#                                 composite set too -- so it is exposed rather
#                                 than hidden behind a fudge factor. Start at
#                                 h_sides=2, h_topbot=1, v_corner=0.4.
#
#                                 The two horizontal terms are both even, so
#                                 they add identically at all four corners and
#                                 can never null one side against the other.
#                                 There is no interaction to tune around.
#
#   convergence_inner   -1..1.5   Shape of the h_sides profile, in edge offsets.
#                                 A real set was converged at the centre AND
#                                 checked at the edge, so what is left over
#                                 peaks somewhere in between -- service
#                                 literature measures inner and outer trap
#                                 error as separate figures for exactly that
#                                 reason. This adds a lobe at half WIDTH worth
#                                 this many edge offsets, and is zero at both
#                                 the centre and the edge, so the calibration
#                                 above is untouched. The profile stops being
#                                 monotonic above 0.47 and the lobe exceeds the
#                                 edge value above 0.71. Because h_sides depends
#                                 on horizontal position alone, the profile is
#                                 exactly what you see scanning along any
#                                 horizontal line -- nothing hides inboard.
#

# ---------------------------------------------------------------------------
# EHT SUPPLY SAG  (stock since 2026-09-01)
# ---------------------------------------------------------------------------
# HISTORY, so the mistake is not repeated. The first attempt measured average
# picture level in a separate shader file, psu-sag.glsl, which SAVEd it as a 1x1
# texture that crt-royale BINDed. That silently does not work: mpv parses each
# entry in glsl-shaders into its own libplacebo hook object and does NOT resolve
# SAVE targets across that boundary. The bind failed, all fifteen passes were
# rejected together, and there was NO error message -- the NTSC stage kept
# running and the CRT stage simply was not there. Do not split a texture
# producer from its consumer across two files in glsl-shaders.
#
# The measurement is a 16th pass inside crt-royale-super.glsl itself. That
# meant renumbering royale_pass_index in every pass and royale_pass_count at its
# six sites, which is the hazard the split was meant to avoid -- but the file
# obeys a checkable invariant (a pass at position p declares index p + 1, and
# count equals the pass total), so it was done and verified programmatically
# rather than by hand, by the generator and merge scripts that built and then
# retired the experimental variant. Both are in git history, not the tree.
#
#
# Declared by crt-royale-super.glsl, so every profile below has it, and it is
# OFF at its default of 0 -- with psu_sag unset these profiles render exactly
# what they rendered before it existed. It is the reason the file is 16 passes
# rather than 15: the meter needs a reduction and a reduction needs its own
# pass. Merged out of the experimental variant on 2026-09-01; see
# shaders/retro/README.md.
#
#   psu_sag             0..1      How badly the anode supply sags under beam
#                                 current. ONE knob on purpose: there is one
#                                 physical quantity (how far the anode voltage
#                                 falls under load) and the raster growth, the
#                                 dimming and the beam limiter are all
#                                 consequences of it, so separate controls
#                                 could only let them contradict each other.
#                                 Anchored at 1 = the repair FAQ's broken set.
#                                 Derived: anode drop d = 0.16*psu_sag, raster
#                                 growth = d/2, dimming = d, beam limiter =
#                                 0.30*sqrt(psu_sag). The limiter is a sqrt
#                                 because EVERY set has one, healthy included,
#                                 so its depth saturates instead of scaling
#                                 with tiredness -- that shape is what lets one
#                                 knob hit both reference sets:
#
#                                   0.125  1% growth, 11% limiter  healthy
#                                   0.25   2%          15%         top of band
#                                   0.375  3%          18%         tired
#                                   1.0    8%          30%         dying
#
#                                 Growth is ISOTROPIC. Real sets bloom more
#                                 horizontally, but that is the per-axis size
#                                 correction the designer fitted, not the
#                                 physics (the patents are explicit that both
#                                 height and width vary), and no source gives a
#                                 ratio -- so there is no H:V knob.
#
#   psu_sag_speed       0.01..1   How fast the BEAM LIMITER settles -- that
#                                 term only. 0.15/frame at 60 Hz is tau ~=
#                                 100 ms, a real limiter time constant. The
#                                 growth and its coupled dimming ignore this
#                                 and snap: the EHT filter capacitance settles
#                                 inside one frame, so at frame granularity the
#                                 raster growth is instantaneous. TYPE DYNAMIC,
#                                 so this one alone retunes with no recompile.
#
# WHERE THE SAG IS APPLIED, since it is not obvious and it is not one place:
#
#   pass  2  meter    averages MAIN_RGB into a 1x1 texture. .r is smoothed over
#                     time and drives the BEAM LIMITER; .g is raw and drives the
#                     RASTER GROWTH and the dimming coupled to it
#   pass  4  droop    applied at the GUN, in the bob pass, in linear light
#   pass  5  droop    again, on DIFFUSION_SRC's de-bob branch, which reads
#                     _PASS_0 directly and would otherwise miss it
#   pass 12  growth   raster lookups scaled at mask-apply, mask left alone
#   pass 13  growth   BLOOM_APPROX lookup scaled, in brightpass
#   pass 15  growth   HALATION_BLUR lookup scaled, in bloom-reconstitute
#
# The last two were missed originally and fixed on 2026-08-05, and the whole
# feature was audited again at the merge on 2026-09-01 -- every pass that
# BINDs a pre-growth texture was checked for whether it actually SAMPLES it,
# resolving #define Source aliases, which is how the first two were missed. The rule for
# deciding which passes need it is mechanical: a texture that DESCENDS from
# MASKED_SCANLINES already carries the growth and is read untouched; a texture
# built from the raster BEFORE mask-apply is pre-growth and its lookup must be
# scaled. Halation is light scattered in the faceplate from the raster's own
# emission -- it cannot sit still while the raster moves, and under kurozumi's
# diffusion_weight that read is most of the visible picture.
#
# Droop has to be at the gun rather than on the finished emission, because
# crt-royale derives beam sigma from the scanline's own intensity via
# beam_spot_power -- so reducing the drive also NARROWS the beam, which is what
# really happens (lower beam current, less space-charge repulsion, tighter
# spot). It is the blooming mechanism running backwards. Applying it later would
# dim without narrowing, and at heavy droop that reads wrong. Halation, diffusion
# and bloom all inherit it too, since they read ORIG_LINEARIZED.
#
# Growth cannot move the same way: mask-apply is the LAST point at which the
# raster and the mask are still separable, and separating them is the whole
# point.
#
# The tweaker has these on page 2 -- press E, then b / s. Drive
# them from the command line too:
#
#   --glsl-shader-opts-append=psu_sag=0.125
#
# ONE key=value PER FLAG. It is not a comma list -- "psu_sag=1,speed=0.3" is
# rejected with "must be a floating point number or a ratio". Repeat the flag:
#
#   --glsl-shader-opts-append=psu_sag=0.125 --glsl-shader-opts-append=psu_sag_speed=0.3
#
# And use -append, never a bare set, or you replace the whole option list.
